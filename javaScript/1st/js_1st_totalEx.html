1식별자

올바른 식별자는 실제로 선언해서 값을 담고, 잘못된 식별자는 eval로 시도해서 어떤 에러가 나는지 직접 확인합니다.

올바른 식별자

let student_ID = "2026001";
let _code = "internal";
let bar = 1;
let Bar = 2; // bar와 Bar는 서로 다른 식별자
RESULT

eval("6variable = 1;"); // 숫자로 시작
eval("if = 1;");        // 예약어 사용
eval("%calc = 1;");     // 허용되지 않는 기호

2문장

    
세미콜론 규칙을 지킨 코드와 지키지 않은 코드를 각각 실행해서 차이를 확인합니다.


    
세미콜론 규칙을 지킨 경우

    
let i = 1;
i = i + 1        // 한 줄 한 문장 → 세미콜론 생략 가능
let k = 1, m = 1;
k = k + 1; m = m + 1; // 한 줄 여러 문장 → 세미콜론 필수

    
RESULT

    


    
세미콜론이 빠져서 에러가 나는 경우

    
eval("let n = 1  let p = 1; n = n + 1  p = p + 1;");
// 첫 번째 문장 끝에 세미콜론이 없어서 SyntaxError 발생

    
RESULT

    

  


  
  

    
3데이터 타입

    
다섯 가지 데이터 타입의 값을 선언하고 typeof로 직접 확인합니다.

    
let num = 42;
let bool = true;
let str = "좋은 세상";
let obj = { name: "루크" };
let empty = null;

typeof num;    typeof bool;   typeof str;
typeof obj;    typeof empty;

    
RESULT

    

  


  
  

    
4변수 (var / let / const)

    
세 가지 선언 방식의 재선언·재할당 가능 여부를 직접 실행해서 비교합니다.


    
var — 재선언도, 재할당도 가능

    
var v = 1;
var v = 2; // 같은 이름으로 재선언해도 에러 없음
v = 3;      // 재할당도 가능

    
RESULT

    


    
let — 재선언 불가, 재할당 가능

    
let score = 0;
score = 10; // (O) 재할당 가능
eval("let score = 20;"); // (X) 같은 스코프에서 재선언 시도 → 에러

    
RESULT

    


    
const — 재선언도 재할당도 불가, 선언 시 초기화 필수

    
const PI = 3.14;
eval("PI = 3.141592;");   // (X) 재할당 시도 → 에러
eval("const NAME;");       // (X) 초기화 없이 선언 → 에러

    
RESULT

    

  


  
  

    
5조건문


    
if / else if / else

    
let hp = 30;

if (hp <= 0) {
  console.log("게임 오버");
} else if (hp < 50) {
  console.log("체력이 위험합니다");
} else {
  console.log("체력이 충분합니다");
}

    
RESULT (hp = 30 / 0 / 80 세 가지 경우를 모두 실행)

    


    
switch

    
let stage = 2;

switch (stage) {
  case 1:
    console.log("튜토리얼");
    break;
  case 2:
    console.log("초급 던전");
    break;
  default:
    console.log("알 수 없는 스테이지");
}

    
RESULT (stage = 1 / 2 / 9 세 가지 경우를 모두 실행)

    

  


  
  

    
6반복문


    
for

    
for (let i = 0; i < 5; i++) {
  console.log("반복 " + i + "번째");
}

    
RESULT

    


    
while

    
let hp = 100;
while (hp > 0) {
  hp -= 20;
  console.log("남은 체력: " + hp);
}

    
RESULT

    


    
do-while

    
let count = 0;
do {
  console.log("count: " + count);
  count++;
} while (count < 3);

    
RESULT

    

  


  
  

    
7break 문과 continue 문


    
break — 반복을 즉시 종료

    
for (let i = 0; i < 10; i++) {
  if (i === 5) {
    break;
  }
  console.log(i);
}

    
RESULT

    


    
continue — 이번 반복만 건너뛰기

    
for (let i = 0; i < 5; i++) {
  if (i === 2) {
    continue;
  }
  console.log(i);
}

    
RESULT
